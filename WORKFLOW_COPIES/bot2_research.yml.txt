name: Bot2 - Research Backtest
on:
  workflow_dispatch:
    inputs:
      download:
        description: Verify/download official Binance archive (does NOT require live API)
        type: boolean
        default: true
permissions:
  contents: write
  actions: write
concurrency:
  group: btc-bot2-v12-state
  cancel-in-progress: false
jobs:
  research:
    runs-on: ubuntu-latest
    timeout-minutes: 120
    steps:
      - uses: actions/checkout@v7.0.1
        with:
          persist-credentials: false
      - uses: actions/setup-python@v7.0.0
        with:
          python-version: '3.13'
          cache: pip
      - run: python -m pip install -r requirements.txt
      - uses: actions/cache/restore@v6.1.0
        with:
          path: data/binance_v12/
          key: binance-v12-${{ hashFiles('config.yaml') }}-${{ github.run_id }}
          restore-keys: |
            binance-v12-${{ hashFiles('config.yaml') }}-
            binance-v12-
      - name: Research on the declared Binance market
        env:
          GITHUB_TOKEN: ${{ github.token }}
          DOWNLOAD: ${{ inputs.download }}
          PYTHONUNBUFFERED: '1'
          OPENBLAS_NUM_THREADS: '1'
        shell: bash
        run: |
          args=(research --output output)
          if [ "$DOWNLOAD" = "true" ]; then args+=(--download); fi
          python main.py "${args[@]}"
      - uses: actions/cache/save@v6.1.0
        if: always()
        with:
          path: data/binance_v12/
          key: binance-v12-${{ hashFiles('config.yaml') }}-${{ github.run_id }}
      - name: Show report
        if: always()
        run: python scripts/show_summary.py
      - uses: actions/upload-artifact@v7.0.1
        if: always()
        with:
          name: bot2-v12-research-${{ github.run_id }}
          path: output/
          retention-days: 90
          if-no-files-found: warn
