name: Bot2 - Hourly Signals
on:
  schedule:
    - cron: '7 * * * *'
  workflow_dispatch:
    inputs:
      mode:
        description: Strict evidence gate; paper is manual research only
        type: choice
        options: [strict, paper, demo, status]
        default: strict
      dry_run:
        description: Do not send Telegram or persist any state
        type: boolean
        default: false
permissions:
  contents: write
concurrency:
  group: btc-bot2-state
  cancel-in-progress: false
jobs:
  scan:
    runs-on: ubuntu-latest
    timeout-minutes: 15
    steps:
      - uses: actions/checkout@v6
        with:
          persist-credentials: false
      - uses: actions/setup-python@v6
        with:
          python-version: '3.13'
          cache: pip
      - run: python -m pip install -r requirements.txt
      - name: Scan completed hourly bars
        env:
          GITHUB_TOKEN: ${{ github.token }}
          GITHUB_REPOSITORY: ${{ github.repository }}
          TELEGRAM_BOT2_TOKEN: ${{ secrets.TELEGRAM_BOT2_TOKEN }}
          TELEGRAM_BOT2_CHAT_ID: ${{ secrets.TELEGRAM_BOT2_CHAT_ID }}
          EVENT: ${{ github.event_name }}
          REQUESTED_MODE: ${{ inputs.mode }}
          DRY_RUN: ${{ inputs.dry_run }}
          PYTHONUNBUFFERED: '1'
        shell: bash
        run: |
          args=(scan --output output)
          if [ "$EVENT" = "workflow_dispatch" ]; then
            args+=(--manual --mode "${REQUESTED_MODE:-strict}")
            if [ "$DRY_RUN" = "true" ]; then args+=(--dry-run); fi
          else
            args+=(--mode strict)
          fi
          python main.py "${args[@]}"
      - name: Show scan status
        if: always()
        shell: bash
        run: |
          if [ -f output/telegram_preview.txt ]; then
            echo '```text' >> "$GITHUB_STEP_SUMMARY"
            cat output/telegram_preview.txt >> "$GITHUB_STEP_SUMMARY"
            echo '' >> "$GITHUB_STEP_SUMMARY"
            echo '```' >> "$GITHUB_STEP_SUMMARY"
          fi
      - uses: actions/upload-artifact@v7
        if: always()
        with:
          name: bot2-scan-${{ github.run_id }}
          path: output/
          retention-days: 7
          if-no-files-found: warn
