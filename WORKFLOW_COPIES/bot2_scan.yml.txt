name: Bot2 - Hourly Signals
on:
  schedule:
    - cron: '7 * * * *'
  workflow_dispatch:
    inputs:
      mode:
        description: Strict needs evidence; paper is unvalidated; demo uses synthetic prices
        type: choice
        options: [strict, paper, demo, status]
        default: status
      dry_run:
        description: Do not send Telegram or update state
        type: boolean
        default: false
permissions:
  contents: write
concurrency:
  group: btc-bot2-v12-state
  cancel-in-progress: false
jobs:
  scan:
    runs-on: ${{ fromJSON(vars.BOT2_LIVE_RUNNER || '["ubuntu-latest"]') }}
    timeout-minutes: 20
    steps:
      - uses: actions/checkout@v7.0.1
        with:
          persist-credentials: false
      - uses: actions/setup-python@v7.0.0
        with:
          python-version: '3.13'
          cache: pip
      - run: python -m pip install -r requirements.txt
      - name: Scan and build risk-defined conditional plans
        env:
          GITHUB_TOKEN: ${{ github.token }}
          TELEGRAM_BOT2_TOKEN: ${{ secrets.TELEGRAM_BOT2_TOKEN }}
          TELEGRAM_BOT2_CHAT_ID: ${{ secrets.TELEGRAM_BOT2_CHAT_ID }}
          EVENT: ${{ github.event_name }}
          REQUESTED_MODE: ${{ inputs.mode }}
          DRY_RUN: ${{ inputs.dry_run }}
          PYTHONUNBUFFERED: '1'
          OPENBLAS_NUM_THREADS: '1'
        shell: bash
        run: |
          args=(scan --output output)
          if [ "$EVENT" = "workflow_dispatch" ]; then
            args+=(--manual --mode "${REQUESTED_MODE:-status}")
            if [ "$DRY_RUN" = "true" ]; then args+=(--dry-run); fi
          else
            args+=(--mode strict)
          fi
          python main.py "${args[@]}"
      - name: Show status
        if: always()
        run: python scripts/show_summary.py
      - uses: actions/upload-artifact@v7.0.1
        if: always()
        with:
          name: bot2-v12-scan-${{ github.run_id }}
          path: output/
          retention-days: 14
          if-no-files-found: warn
